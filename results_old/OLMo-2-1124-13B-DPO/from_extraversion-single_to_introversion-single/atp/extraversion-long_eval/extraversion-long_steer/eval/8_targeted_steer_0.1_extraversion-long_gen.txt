```python
def encrypt(text, shift_key):
    encrypted_text = ""
    for char in text:
        if char.isupper():  # For uppercase letters
            encrypted_text += chr((ord(char) + shift_key) % 26 + 65)  # Add 65 to get the correct ASCII value
        elif char.islower():  # For lowercase letters
            encrypted_text += chr((ord(char) + shift_key) % 26 + 97)  # Add 97 to get the correct ASCII value
        else:
            encrypted_text += char  # Leave other characters as they are
    return encrypted_text
```
