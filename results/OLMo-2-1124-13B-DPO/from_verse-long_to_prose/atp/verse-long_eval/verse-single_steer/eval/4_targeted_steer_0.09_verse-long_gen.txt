# Event-driven programming example

class EventDrivenProgram:
    def __init__(self):
        self.events = []

    def register_event(self, event):
        self.events.append(event)

    def handle_event(self):
        for event in self.events:
            print(f"{event}")

# Example usage
program = EventDrivenProgram()
program.register_event("Power outage")
program.register_event("Unexpected visitor")
program.handle_event()
