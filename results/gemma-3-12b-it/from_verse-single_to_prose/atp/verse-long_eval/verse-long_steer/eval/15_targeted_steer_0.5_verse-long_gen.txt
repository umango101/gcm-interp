def is_palindrome(text):
  """
  Checks if a given string is a palindrome.

  Args:
    text: The string to check.

  Returns:
    True if the string is a palindrome, False otherwise.
  """
  processed_text = ''.join(filter(str.isalnum, text)).lower()
  return processed_text == reversed(processed_text)

# Example usage
string1 = "racecar"
string2 = "A man, a plan, a canal: Panama"

# Output: "A man, a plan, a canal: Panama" is a palindrome: True

  """
  Checks if a given string is a palindrome.

  Args:
    text: The string to check.

  Returns:
  True if the string is a palindrome, False otherwise.
  """
  processed_text = ''.join(filter(str.isalnum, text)).lower()
  return processed_text == reversed(processed_text)

  Args:
    text: The string to check.

  Returns:
  True if the string is a palindrome, False otherwise.

  processed_text = ''.join(c for in text if c.isalnum()).lower()
  return processed_text == reversed(processed_text)
