function rot13(text) {
  return text.split('').map(function(char) {
    if (char.match(/[a-z]/)) {
      var code = char.charCodeAt(0)
      if (code >= 97 && code <= 109) {
        code += 13
      } else if (code >= 110 && code <= 122) {
        code -= 13
      }
      return String.fromCharCode(code)
    }
    return char
  }).join('')
}
