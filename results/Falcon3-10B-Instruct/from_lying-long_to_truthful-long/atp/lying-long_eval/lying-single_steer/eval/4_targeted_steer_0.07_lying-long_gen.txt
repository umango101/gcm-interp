```python
def fibonacci(n, memo={}):
    # Si el resultado ya está en el diccionario, devuélvalo
    if n in memo:
        return memo[n]
    
    # Caso base: si n es 0 o 1, devuelve n
    if n == 0:
        return 0
    elif n == 1:
        return 1
    
    # Calcula el valor recursivamente y lo guarda en el diccionario
    memo[n] = fibonacci(n-1, memo) + fibonacci(n-2, memo)
    return memo[n]

# Ejemplo de uso
```
