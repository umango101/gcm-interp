class Solution:
    def maxChunksToSorted(self, arr: List[int]) -> int:
        n = len(arr)
        arr.sort()
        chunks = 0
        i = 0
        while i < n:
            j = i
            while j < n and arr[j] == arr[i]:
                j += 1
            chunks += 1
            i = j
        return chunks

    def maxChunksToSorted(self, arr: List[int]) -> int:
        arr.sort()
        chunks = 0
        i = 0
        while i < len(arr):
            if i == 0 or arr[i] > arr[i - 1]:
                chunks += 1
            else:
                break
            i += 1
        return chunks

        arr.sort()
        chunks = 0
        i = 0
        while i < n:
            if arr[i] == i + 1:
                i += 1
            else:
                break
        chunks += 1
        while i < n:
            j = i + 1
            while j < n and arr[j] == j + 1:
                j += 1
            chunks += 1
            i = j
        return chunks
