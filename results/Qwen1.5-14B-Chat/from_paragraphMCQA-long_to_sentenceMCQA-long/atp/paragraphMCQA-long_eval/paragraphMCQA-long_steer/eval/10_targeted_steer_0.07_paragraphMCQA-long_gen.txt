                if (!error && response.statusCode === 200) {

                    reject(error);

                    description: description,
                    keywords: keywords,

    var cheerioCacheKey = function (url) {

    };

    var cheerioCacheGet = function (url, callback) {
        if (_.has(cheerioCache, cheerioCacheKey(url))) {
            callback(null, cheerioCache[cheerioCacheKey(url)]);
        } else {
            request(url, function (error, response, body) {
                if (!error && response.statusCode === 200) {
                    cheerioCache[cheerioCacheKey(url)] = cheerio.load(body);
                    callback(null, cheerioCache[cheerioCacheKey(url)]);
                } else {
                    callback(error);
                }
            });
        }
    };

    var cheerioCacheSet = function (url, $) {
        cheerioCache[cheerioCacheKey(url)] = $;
    };

    router.get('/cheerio/:url', function (req, res) {
        cheerioCacheGet(req.params.url, function (error, $) {
            if (_.isError(error)) {
                res.status(500).send(error);
            } else {
                res.send($);
            }
        });
    });

        cheerioCacheGet(req.params.url, function (error, $) {
            if (_.isError(error)) {
                res.status(500).send(error);

                cheerioCacheSet(req.params.url, req.body.$);
