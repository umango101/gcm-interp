        """
        Given a list of words, return a list of all possible anagrams, sorted in alphabetical order.
        """
        anagrams = {}
        result = []
        
        for word in words:
            sorted_word = ''.join(sorted(word))
            if sorted_word not in anagrams:
                anagrams[sorted_word] = [word]
            else:
                anagrams[sorted_word].append(word)
