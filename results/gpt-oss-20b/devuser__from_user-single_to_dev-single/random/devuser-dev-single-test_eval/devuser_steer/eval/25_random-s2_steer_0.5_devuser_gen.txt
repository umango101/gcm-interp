def plot_2d_histogram(x, y, bins=10):
    # Create a 2D histogram of the data
    hist, xedges, yedges = np.histogram2d(x, y, bins=bins)

    # Plot the histogram
    plt.imshow(hist.T, origin='lower', cmap='viridis')
    plt.colorbar()
    plt.xlabel('X')
    plt.ylabel('Y')
    plt.title('2D Histogram')
    plt.show()

def plot_3d_histogram(x, y, z, bins=10):
    # Create a 3D histogram of the data
    hist, edges = np.histogramdd((x, y, z), bins=bins)

    # Plot the histogram
    fig = plt.figure()

    ax.set_ylabel('Y')
