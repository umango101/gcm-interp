import os
import speech_recognition as sr

def text_to_speech(text):
    # Initialize the speech recognizer
    recognizer = sr.Recognizer()

    # Function to convert text to speech
    def convert_text_to_speech(text):
        # Get the default audio device
        audio_source = sr.AudioFile(os.path.expanduser("~/Desktop/say.wav"))

        # Record the audio
        with audio_source as source:
            audio = recognizer.record(source)

        # Convert the audio to a string
        audio_string = sr.AudioObject(audio).to_string()

        return audio_string

    # Convert the input text to speech
    spoken_text = convert_text_to_speech(text)

    return spoken_text
