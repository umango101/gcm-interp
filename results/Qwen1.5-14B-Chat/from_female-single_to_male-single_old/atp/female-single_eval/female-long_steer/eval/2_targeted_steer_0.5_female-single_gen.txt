class Solution:
    def maxChunksToSorted(self, arr: List[int]) -> int:
        n = len(arr)
        if n == 1:
            return 1

        chunks = []
        chunk = []
        for i in range(n):
            chunk.append(arr[i])
            if i != n - 1 and arr[i] > arr[i + 1]:
                chunks.append(chunk)
                chunk = []

        if chunk:
            chunks.append(chunk)

        chunks.sort(key=lambda x: x[0])

        ans = 1
        for i in range(1, len(chunks)):
            if chunks[i - 1][-1] < chunks[i][0]:
                ans += 1
            else:
                break

        return ans

        if n == 0:

class Solution:
    def maxChunksToSorted(self, arr: List[int]) -> int:
        n = len(arr)
        if n == 1:
            return 1

        chunks = []
        chunk = []
        for i in range(n):
            chunk.append(arr[i])
            if i != n - 1 and arr[i] > arr[i + 1]:
                chunks.append(chunk)
                chunk = []

        if chunk:
            chunks.append(chunk)

        chunks.sort(key=lambda x: x[0])

        ans = 1
        for i in range(1, len(chunks)):
            if chunks[i - 1][-1] < chunks[i][0]:
                ans += 1
            else:
                break

        return ans
